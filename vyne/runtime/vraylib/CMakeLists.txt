cmake_minimum_required(VERSION 3.20)
project(vyne_raylib C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPENGL_VERSION "3.3" CACHE STRING "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_X11 ON CACHE BOOL "" FORCE)
set(USE_EXTERNAL_GLFW OFF CACHE STRING "" FORCE)
add_subdirectory("../../../vendor/raylib" raylib)
# vimage compiles its own stb implementations in the user's translation unit.
# Keep raylib's versions private instead of exporting duplicate stb symbols.
target_compile_definitions(raylib PRIVATE STB_IMAGE_STATIC STB_IMAGE_WRITE_STATIC SUPPORT_FILEFORMAT_JPG=1)
add_library(vyne_raylib STATIC bridge.c)
target_include_directories(vyne_raylib PRIVATE "../../../vendor/raylib/src")
target_compile_definitions(vyne_raylib PRIVATE GRAPHICS_API_OPENGL_33)
target_link_libraries(vyne_raylib PUBLIC raylib)
set_target_properties(vyne_raylib raylib PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(vyne_raylib raylib PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
